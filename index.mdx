export const TryIt = () => {
  const [active, setActive] = useState(0);
  const item = TRY_ITEMS[active];
  return (
    <div className="hr-try">
      <div className="hr-try-list" role="tablist">
        {TRY_ITEMS.map((it, i) => (
          <button
            key={it.label}
            role="tab"
            aria-selected={i === active}
            className={i === active ? "hr-try-item hr-try-item-active" : "hr-try-item"}
            onClick={() => setActive(i)}
          >
            {it.label}
          </button>
        ))}
      </div>
      <div className="hr-code hr-try-code">
        <div className="hr-code-bar">
          <span style={{ display: "flex", alignItems: "center" }}>
            <span className="hr-dots"><span /><span /><span /></span>
            <span className="hr-method">{item.method}</span>{item.path}
          </span>
          <span>test mode</span>
        </div>
        <div className="hr-pre">
          {item.code.split("\n").map((line, i) => (
            <div key={i} className="hr-line">{line}</div>
          ))}
        </div>
        <div className="hr-try-foot">
          <a href={item.href} className="hr-link">Open in the API playground →</a>
        </div>

    </div>
  );
};
